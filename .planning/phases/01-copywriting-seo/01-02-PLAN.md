**Zod schema (define at top of file):**
```typescript
import { z } from 'zod';
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { toast } from 'sonner';

const contactSchema = z.object({
  name: z.string().min(2, 'Votre prenom est requis'),
  email: z.string().email('Adresse email invalide'),
  sector: z.enum(['conseil', 'formation', 'services-pro', 'autre'], {
    required_error: "Selectionnez votre secteur d'activite",
  }),
  problem: z.string().min(20, 'Decrivez votre situation en quelques mots (min. 20 caracteres)'),
});

type ContactFormData = z.infer<typeof contactSchema>;
```
